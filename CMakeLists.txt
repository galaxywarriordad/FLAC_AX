cmake_minimum_required(VERSION 3.25)
project(FLAC_AX VERSION 1.36.2 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(FLAC_AX_BUILD_STUDIO "Build ax_av_studio live synth + OpenGL HUD" ON)
option(FLAC_AX_BUILD_UPSCALER "Build ax_video_upscaler DVD/HDMI video upscaler" ON)
option(FLAC_AX_BUILD_APPLIANCE_TUI "Build ax_appliance_tui Core 5 appliance launcher" ON)
option(FLAC_AX_BUILD_ENCODE_2GO "Build ax_encode_2go BATLEXP G350 video encoder launcher" ON)
option(FLAC_AX_BUILD_LV2_HOST "Build ax_lv2_host single-plugin LV2 runner when lilv-0 is available" ON)
option(FLAC_AX_BUILD_CLAP_HOST "Build ax_clap_host single-plugin CLAP runner when clap/clap.h is available" ON)
option(FLAC_AX_USE_AX_SDK "Link ax_core (ax-sdk) for canonical 2:3:1 / Core 5 topology pinning" ON)
option(FLAC_AX_ENABLE_NATIVE_OPT "Enable -march=skylake / native tuning when supported" ON)
option(FLAC_AX_ENABLE_LTO "Enable Interprocedural Optimization (LTO) in Release builds" ON)
option(FLAC_AX_ENABLE_WARNINGS "Enable compiler warnings for FLAC_AX targets" ON)

if(FLAC_AX_USE_AX_SDK)
    include(FetchContent)
    set(AX_SDK_LOCAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../ax-sdk")
    if(EXISTS "${AX_SDK_LOCAL_DIR}/CMakeLists.txt")
        message(STATUS "FLAC_AX: Using local ax-sdk at ${AX_SDK_LOCAL_DIR}")
        add_subdirectory("${AX_SDK_LOCAL_DIR}" "${CMAKE_BINARY_DIR}/_deps/ax_sdk-build" EXCLUDE_FROM_ALL)
    else()
        message(STATUS "FLAC_AX: Fetching ax-sdk from GitHub (main)")
        FetchContent_Declare(
            ax_sdk
            GIT_REPOSITORY https://github.com/galaxywarriordad/ax-sdk.git
            GIT_TAG main
            GIT_SHALLOW TRUE
        )
        FetchContent_MakeAvailable(ax_sdk)
    endif()

    set(_flac_ax_clang_helper "${AX_SDK_LOCAL_DIR}/ax_toolchain/piccolo-clang-x86_64.cmake")
    if(NOT EXISTS "${_flac_ax_clang_helper}" AND DEFINED ax_sdk_SOURCE_DIR)
        set(_flac_ax_clang_helper "${ax_sdk_SOURCE_DIR}/ax_toolchain/piccolo-clang-x86_64.cmake")
    endif()
    if(EXISTS "${_flac_ax_clang_helper}")
        include("${_flac_ax_clang_helper}")
    endif()
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    if(MSVC OR CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
        add_compile_definitions(_ALLOW_COMPILER_AND_STL_VERSION_MISMATCH=1 _DISABLE_VECTOR_ANNOTATION)
        if(FLAC_AX_ENABLE_NATIVE_OPT)
            add_compile_options(/clang:-O3 /clang:-march=skylake /clang:-mbmi /clang:-mbmi2)
        endif()
    else()
        if(FLAC_AX_ENABLE_NATIVE_OPT)
            add_compile_options(-O3 -march=skylake -mbmi -mbmi2)
        endif()
    endif()
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND FLAC_AX_ENABLE_NATIVE_OPT AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    add_compile_options(-O3 -march=skylake -mbmi -mbmi2)
endif()

if(FLAC_AX_ENABLE_LTO)
    include(CheckIPOSupported)
    check_ipo_supported(RESULT _flac_ax_ipo_ok OUTPUT _flac_ax_ipo_err)
    if(_flac_ax_ipo_ok)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    else()
        message(STATUS "FLAC_AX: IPO/LTO not enabled (${_flac_ax_ipo_err})")
    endif()
endif()

endif()

function(flac_ax_configure_target target_name)
    target_compile_features("${target_name}" PRIVATE cxx_std_20)
    target_include_directories("${target_name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    if(MSVC OR CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
        target_compile_definitions("${target_name}" PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
    if(FLAC_AX_ENABLE_WARNINGS)
        if(MSVC OR CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
            if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
                target_compile_options("${target_name}" PRIVATE /W4 /clang:-Wno-c++98-compat)
            else()
                target_compile_options("${target_name}" PRIVATE /W4)
            endif()
        else()
            target_compile_options("${target_name}" PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
    if(UNIX)
        target_link_libraries("${target_name}" PRIVATE pthread dl)
    endif()
    if(FLAC_AX_USE_AX_SDK AND TARGET ax_core)
        target_link_libraries("${target_name}" PRIVATE ax_core)
    endif()
endfunction()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
    set(FLAC_AX_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/tracker_openmpt_reader.cpp")
        list(APPEND FLAC_AX_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/tracker_openmpt_reader.cpp")
    endif()
    add_executable(flac_ax ${FLAC_AX_SOURCES})
    if(FLACPP_FOUND)
        target_include_directories(flac_ax PRIVATE ${FLACPP_INCLUDE_DIRS})
        target_link_directories(flac_ax PRIVATE ${FLACPP_LIBRARY_DIRS})
        target_link_libraries(flac_ax PRIVATE ${FLACPP_LIBRARIES})
        target_compile_definitions(flac_ax PRIVATE FLAC_AX_HAS_FLACPP=1)
    elseif(UNIX)
        target_link_libraries(flac_ax PRIVATE FLAC++ FLAC)
        target_compile_definitions(flac_ax PRIVATE FLAC_AX_HAS_FLACPP=1)
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/flac/include/FLAC++/decoder.h")
        target_include_directories(flac_ax PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/flac/include")
        target_compile_definitions(flac_ax PRIVATE FLAC_AX_HAS_FLACPP=1)
    endif()
    if(ALSA_FOUND)
        target_include_directories(flac_ax PRIVATE ${ALSA_INCLUDE_DIRS})
        target_link_directories(flac_ax PRIVATE ${ALSA_LIBRARY_DIRS})
        target_link_libraries(flac_ax PRIVATE ${ALSA_LIBRARIES})
    elseif(UNIX)
        target_link_libraries(flac_ax PRIVATE asound)
    endif()
    flac_ax_configure_target(flac_ax)
endif()

if(FLAC_AX_BUILD_CLAP_HOST AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/hosts/ax_clap_host.cpp")
    find_path(CLAP_HEADER_DIR NAMES clap/clap.h)
    if(NOT CLAP_HEADER_DIR)
        include(FetchContent)
        FetchContent_Declare(
            clap_headers
            GIT_REPOSITORY https://github.com/free-audio/clap.git
            GIT_TAG main
            GIT_SHALLOW TRUE
        )
        FetchContent_GetProperties(clap_headers)
        if(NOT clap_headers_POPULATED)
            FetchContent_Populate(clap_headers)
        endif()
        if(EXISTS "${clap_headers_SOURCE_DIR}/include/clap/clap.h")
            set(CLAP_HEADER_DIR "${clap_headers_SOURCE_DIR}/include")
        endif()
    endif()

    if(CLAP_HEADER_DIR)
        add_executable(ax_clap_host src/hosts/ax_clap_host.cpp)
        target_include_directories(ax_clap_host PRIVATE "${CLAP_HEADER_DIR}")
        flac_ax_configure_target(ax_clap_host)
        message(STATUS "FLAC_AX: Configured ax_clap_host target using ${CLAP_HEADER_DIR}")
    else()
        message(STATUS "FLAC_AX: clap/clap.h not found; skipping ax_clap_host")
    endif()
endif()
